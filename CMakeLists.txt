cmake_minimum_required(VERSION 3.13)
set(PICO_PLATFORM rp2350-arm-s CACHE STRING "rp2040, rp2350-arm-s or rp2350-riscv")
if(NOT DEFINED PICO_BOARD)
 if(PICO_PLATFORM STREQUAL "rp2040")
  set(PICO_BOARD pico CACHE STRING "Board")
 else()
  set(PICO_BOARD pico2 CACHE STRING "Board")
 endif()
endif()
include(pico_sdk_import.cmake)
project(mimiboot_night C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
set(MB_ROOT ${CMAKE_CURRENT_LIST_DIR})
set(MB_CORE core/mb_hash.c core/mb_ed25519.c core/mb_image.c core/mb_bcl.c core/mb_config.c core/mb_policy.c core/mb_recv.c core/mb_console.c core/mb_control.c core/mb_grub.c core/mb_formats.c)
if(PICO_PLATFORM STREQUAL "rp2350-riscv")
 list(APPEND MB_CORE port/rp2350/rv_helpers.S port/rp2350/rv_div64.c)
endif()
option(MIMIBOOT_DIAGNOSTICS "Retain temporary boot progress for bring-up" OFF)
option(MIMIBOOT_UNIVERSAL "RP2350 half of a universal Arm/RISC-V loader" OFF)
option(MIMIBOOT_REQUIRE_SIGNATURE "Refuse unsigned images" OFF)
option(MIMIBOOT_RAM_PROBE "Also build the write-locked RP2040 RAM probe" ON)
set(MIMIBOOT_CONFIG_HEADER "" CACHE FILEPATH "mimiconf-generated feature header (empty: every Night feature on)")
set(MIMIBOOT_FLASH_SIZE "" CACHE STRING "Override the board's flash size in bytes (selects the 2/4/16 MB layout)")
if(MIMIBOOT_REQUIRE_SIGNATURE AND NOT DEFINED MIMIBOOT_KEYS)
 message(FATAL_ERROR "MIMIBOOT_REQUIRE_SIGNATURE requires MIMIBOOT_KEYS")
endif()
function(night_common target)
 if(MIMIBOOT_DIAGNOSTICS)
  target_compile_definitions(${target} PRIVATE MB_BOOT_DIAGNOSTICS=1)
 endif()
 target_include_directories(${target} PRIVATE . include)
 target_compile_options(${target} PRIVATE -Os -Wall -Wextra -Wno-unused-parameter -fstack-usage)
 target_compile_definitions(${target} PRIVATE PICO_STACK_SIZE=0x2000)
 if(DEFINED MIMIBOOT_KEYS)
  get_filename_component(keys ${MIMIBOOT_KEYS} ABSOLUTE)
  target_compile_definitions(${target} PRIVATE MB_KEYS_HEADER="${keys}")
 endif()
 if(MIMIBOOT_REQUIRE_SIGNATURE)
  target_compile_definitions(${target} PRIVATE MB_REQUIRE_SIGNATURE=1)
 endif()
 if(MIMIBOOT_UNIVERSAL)
  target_compile_definitions(${target} PRIVATE MB_RP2350_UNIVERSAL=1)
 endif()
 if(MIMIBOOT_CONFIG_HEADER)
  get_filename_component(cfgh ${MIMIBOOT_CONFIG_HEADER} ABSOLUTE)
  target_compile_definitions(${target} PRIVATE MB_CONFIG_HEADER="${cfgh}")
 endif()
 if(MIMIBOOT_FLASH_SIZE)
  target_compile_definitions(${target} PRIVATE PICO_FLASH_SIZE_BYTES=${MIMIBOOT_FLASH_SIZE})
 endif()
 pico_enable_stdio_usb(${target} 1)
 pico_enable_stdio_uart(${target} 0)
 pico_set_program_name(${target} "MimiBoot-Night")
 pico_set_program_version(${target} "2.0.0 Night")
 pico_add_extra_outputs(${target})
endfunction()
# Loader uses only core0. Combine scratch banks into an 8 KiB stack for
# nested signature verification and console/service calls.
include(${MB_ROOT}/tools/night_linker.cmake)
if(PICO_PLATFORM STREQUAL "rp2040")
 file(READ ${PICO_SDK_PATH}/src/rp2_common/pico_crt0/rp2040/memmap_default.ld ld)
 string(REPLACE "RAM(rwx) : ORIGIN =  0x20000000, LENGTH = 256k" "RAM(rwx) : ORIGIN =  0x20020000, LENGTH = 128k" ld "${ld}")
 night_stack(ld)
 file(WRITE ${CMAKE_BINARY_DIR}/loader.ld "${ld}")
 add_executable(mimiboot_night ${MB_CORE} port/rp2040/mb_rp2040.c)
 pico_set_linker_script(mimiboot_night ${CMAKE_BINARY_DIR}/loader.ld)
 target_link_libraries(mimiboot_night pico_stdlib pico_stdio_usb hardware_flash hardware_watchdog hardware_sync hardware_irq)
 night_common(mimiboot_night)
 if(MIMIBOOT_RAM_PROBE)
  file(READ ${PICO_SDK_PATH}/src/rp2_common/pico_crt0/rp2040/memmap_no_flash.ld ram_ld)
  string(REPLACE "RAM(rwx) : ORIGIN =  0x20000000, LENGTH = 256k" "RAM(rwx) : ORIGIN =  0x20010000, LENGTH = 192k" ram_ld "${ram_ld}")
  night_stack(ram_ld)
  file(WRITE ${CMAKE_BINARY_DIR}/probe.ld "${ram_ld}")
  add_executable(mimiboot_night_ram ${MB_CORE} port/rp2040/mb_rp2040.c)
  pico_set_binary_type(mimiboot_night_ram no_flash)
  pico_set_linker_script(mimiboot_night_ram ${CMAKE_BINARY_DIR}/probe.ld)
  target_compile_definitions(mimiboot_night_ram PRIVATE MB_RAM_PROBE=1)
  target_link_libraries(mimiboot_night_ram pico_stdlib pico_stdio_usb hardware_flash hardware_watchdog hardware_sync hardware_irq)
  night_common(mimiboot_night_ram)
 endif()
else()
 file(READ ${PICO_SDK_PATH}/src/rp2_common/pico_crt0/rp2350/memmap_default.ld ld)
 string(REPLACE "RAM(rwx) : ORIGIN =  0x20000000, LENGTH = 512k" "RAM(rwx) : ORIGIN =  0x20070000, LENGTH = 64k" ld "${ld}")
 if(NOT ld MATCHES "0x20070000")
  message(FATAL_ERROR "SDK memory map changed")
 endif()
 night_stack(ld)
 file(WRITE ${CMAKE_BINARY_DIR}/loader.ld "${ld}")
 add_executable(mimiboot_night ${MB_CORE} port/rp2350/mb_rp2350.c port/rp2350/main.c)
 pico_set_linker_script(mimiboot_night ${CMAKE_BINARY_DIR}/loader.ld)
 target_link_libraries(mimiboot_night pico_stdlib pico_bootrom pico_sha256 hardware_flash hardware_watchdog hardware_resets hardware_irq)
 night_common(mimiboot_night)
endif()
add_custom_command(TARGET mimiboot_night POST_BUILD COMMAND ${CMAKE_COMMAND} -DBIN=$<TARGET_FILE_DIR:mimiboot_night>/mimiboot_night.bin -DLIMIT=245760 -P ${MB_ROOT}/tools/size_check.cmake)
