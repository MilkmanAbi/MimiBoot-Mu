cmake_minimum_required(VERSION 3.13)
set(MB_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
include(${MB_ROOT}/pico_sdk_import.cmake)
project(rp_client C CXX ASM)
pico_sdk_init()
# This example never starts core1. SMP applications must leave this OFF and
# supply a safe flash rendezvous on both cores.
set(MIMIBOOT_CLIENT_SINGLE_CORE ON CACHE BOOL "Example is single core")
include(${MB_ROOT}/mimiboot_client.cmake)
include(${MB_ROOT}/tools/night_linker.cmake)
add_executable(rp_client main.c)
target_link_libraries(rp_client pico_stdlib mimiboot_client)
target_compile_definitions(rp_client PRIVATE PICO_STACK_SIZE=0x2000)
if(PICO_PLATFORM STREQUAL "rp2040")
 pico_set_binary_type(rp_client no_flash)
 file(READ ${PICO_SDK_PATH}/src/rp2_common/pico_crt0/rp2040/memmap_no_flash.ld ld)
 # Keep payload and BSS in the lower 128 KiB. Loader is alive while staging it.
 string(REPLACE "LENGTH = 256k" "LENGTH = 128k" ld "${ld}")
else()
 file(READ ${PICO_SDK_PATH}/src/rp2_common/pico_crt0/rp2350/memmap_default.ld ld)
endif()
night_stack(ld)
file(WRITE ${CMAKE_BINARY_DIR}/client.ld "${ld}")
pico_set_linker_script(rp_client ${CMAKE_BINARY_DIR}/client.ld)
pico_enable_stdio_usb(rp_client 1)
pico_enable_stdio_uart(rp_client 0)
pico_add_extra_outputs(rp_client)
